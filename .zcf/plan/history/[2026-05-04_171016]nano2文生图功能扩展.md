#### 步骤 1.4 - 创建适配器工厂

**文件**：`src/lib/api/adapters/index.ts`

```typescript
import { NanoBanana2Adapter } from './NanoBanana2Adapter'
import { NanoBananaProAdapter } from './NanoBananaProAdapter'
import { GPTImageAdapter } from './GPTImageAdapter'
import type { ImageGeneratorAdapter } from './ImageAdapter'

export const IMAGE_ADAPTERS: Record<ImageModelId, ImageGeneratorAdapter> = {
  'nano-banana2': new NanoBanana2Adapter(),
  'nano-banana-pro': new NanoBananaProAdapter(),
  'gpt-image-2': new GPTImageAdapter(),
}

export function getAdapter(modelId: ImageModelId): ImageGeneratorAdapter {
  return IMAGE_ADAPTERS[modelId]
}
```

**预期结果**：通过 `getAdapter(modelId)` 获取对应模型适配器
